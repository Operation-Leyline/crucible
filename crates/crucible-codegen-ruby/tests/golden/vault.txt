module Vault
  # Client for the Vault service.
  #
  # Stores secrets.
  #
  # See [the guide](https://example.com/guide) and \[Secret] for `*/` details. @since 1
  #
  # Create one `VaultClient` and share it: it is safe to call from many threads at once, and it keeps its HTTP connection open between calls. Call {#close} when finished. Every operation blocks until it completes and reports failure as a `VaultException` for a service response, a {Runtime::NetworkError} when no response arrives, a {Runtime::DeadlineError} when the call outlasts {Runtime::ClientConfig#timeout}, or a {Runtime::CancelledError}.
  class VaultClient
    # Creates a client. Without a `config`, the generated endpoint is used and no credentials are sent; see {Runtime::ClientConfig} for credentials, retries, timeouts and the transport.
    #
    # @param config [Runtime::ClientConfig]
    # @raise [ArgumentError] when the endpoint or another setting is invalid
    def initialize(config = Runtime::ClientConfig.new)
      @runtime = Runtime::ClientRuntime.new(config, ::Vault.codec, endpoint: '', digest: false)
    end
    # Releases the HTTP connection this client created. A transport passed in the configuration is left open.
    def close = @runtime.close
    # Reads one secret.
    #
    # Read-only: retried automatically after transient failures.
    #
    # @param input [GetSecretInput] the request to send
    # @param cancellation [Runtime::CancellationToken, nil] cancels the call when cancelled
    # @return [GetSecretOutput] the service's response
    # @raise [AccessDenied]
    # @raise [Runtime::VaultException] for any other error response
    def get_secret(input, cancellation: nil)
      @runtime.call('GetSecret', input, cancellation: cancellation)
    end
    # Lists secrets.
    #
    # Read-only: retried automatically after transient failures.
    #
    # This operation is paginated: use the `_pages` variant to read every page.
    #
    # @param input [ListSecretsInput] the request to send
    # @param cancellation [Runtime::CancellationToken, nil] cancels the call when cancelled
    # @return [ListSecretsOutput] the service's response
    # @raise [AccessDenied]
    # @raise [Runtime::VaultException] for any other error response
    def list_secrets(input, cancellation: nil)
      @runtime.call('ListSecrets', input, cancellation: cancellation)
    end
    # Every page of {#list_secrets}, following the pagination token until the service stops returning one. Returns an `Enumerator`; the call for each page is made as it is consumed, so each can fail.
    #
    # @param input [ListSecretsInput] the request for the first page
    # @param cancellation [Runtime::CancellationToken, nil]
    # @return [Enumerator<ListSecretsOutput>]
    def list_secrets_pages(input, cancellation: nil)
      @runtime.pages('ListSecrets', input, cancellation: cancellation)
    end
    # Stores a secret.
    #
    # Idempotent: retried automatically after transient failures.
    #
    # `client_token` is filled with a fresh token when omitted, so retries do not repeat the effect.
    #
    # @param input [PutSecretInput] the request to send
    # @param cancellation [Runtime::CancellationToken, nil] cancels the call when cancelled
    # @return [PutSecretOutput] the service's response
    # @raise [AccessDenied]
    # @raise [Runtime::VaultException] for any other error response
    def put_secret(input, cancellation: nil)
      @runtime.call('PutSecret', input, cancellation: cancellation)
    end
  end
end

        when 'username' then UsernameVariant.new(value.nil? ? nil : context.codec.decode(value, context.member(key)))

        end
