<?php
declare(strict_types=1);
namespace VaultSdk;

/**
 * Client for the Vault service.
 *
 * Stores secrets.
 *
 * See [the guide](https://example.com/guide) and {@see \VaultSdk\Secret} for `*\/` details. @since 1
 *
 * Create one `VaultClient` per configuration and reuse it: it keeps its HTTP connections open between calls. A client is not safe to share across threads or fibers that run calls at the same time; give each its own. Call `close()` when finished. Every operation blocks until it completes and reports failure as a ServiceException for a service response, a NetworkException when no response arrives, a TimeoutException when the call outlasts `ClientConfig::$timeout`, or a CancelledException.
 */
final class VaultClient
{
    private readonly Runtime\ClientRuntime $runtime;
    /**
     * Creates a client. Without a `$config`, the generated endpoint is used and no credentials are sent; see ClientConfig for credentials, retries, timeouts and the transport.
     *
     * @throws \InvalidArgumentException when the endpoint or another setting is invalid
     */
    public function __construct(Runtime\ClientConfig $config = new Runtime\ClientConfig())
    {
        $this->runtime = new Runtime\ClientRuntime($config, Schema::codec(), Decoder::error(...), '');
    }
    /** Releases the HTTP transport this client created. A transport passed in the configuration is left open. */
    public function close(): void { $this->runtime->close(); }
    /**
     * Reads one secret.
     *
     * Read-only: retried automatically after transient failures.
     *
     * @param \VaultSdk\GetSecretInput $input the request to send
     * @param ?Runtime\CancellationToken $cancellation cancels the call when cancelled
     * @return \VaultSdk\GetSecretOutput the service's response
     * @throws \VaultSdk\AccessDenied
     * @throws \VaultSdk\VaultException for any other error response
     */
    public function getSecret(\VaultSdk\GetSecretInput $input, ?Runtime\CancellationToken $cancellation = null): \VaultSdk\GetSecretOutput
    {
        return $this->runtime->call('GetSecret', $input, $cancellation);
    }
    /**
