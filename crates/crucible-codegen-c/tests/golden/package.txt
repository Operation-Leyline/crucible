cmake_minimum_required(VERSION 3.20)
project(weather_client VERSION 0.0.0 LANGUAGES C)
find_package(yyjson CONFIG REQUIRED)
find_package(CURL 7.68 REQUIRED)
find_package(OpenSSL 1.1.1 REQUIRED)
find_package(Threads REQUIRED)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(weather_client STATIC src/models.c src/client.c src/arena.c src/json.c src/number.c src/timestamp.c src/codec.c src/scope.c src/headers.c src/transport.c src/client_runtime.c src/bindings.c)
add_library(weather_client::weather_client ALIAS weather_client)
target_compile_features(weather_client PUBLIC c_std_17)
set_target_properties(weather_client PROPERTIES C_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON DEBUG_POSTFIX d)
target_link_libraries(weather_client PRIVATE yyjson::yyjson CURL::libcurl OpenSSL::Crypto Threads::Threads)
target_include_directories(weather_client PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
if(MSVC)
  target_compile_definitions(weather_client PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(weather_client PRIVATE /W4 /WX)
else()
  target_compile_options(weather_client PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(weather_client PRIVATE m)
endif()
install(TARGETS weather_client EXPORT weather_clientTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT weather_clientTargets NAMESPACE weather_client:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/weather_client)
configure_package_config_file(cmake/Config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/weather_clientConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/weather_client)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/weather_clientConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/weather_clientConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/weather_clientConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/weather_client)
install(FILES crucible.json schema.json DESTINATION ${CMAKE_INSTALL_DATADIR}/weather_client)
